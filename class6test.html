import cv2 import numpy as np import math # Video Settings WIDTH, HEIGHT = 3840, 2160 # 4K Resolution FPS = 60 DURATION_SECONDS = 10 # Loop duration OUTPUT_FILE = "blackhole_website_bg.mp4" # Initialize Video Writer fourcc = cv2.VideoWriter_fourcc(*'mp4v') out = cv2.VideoWriter(OUTPUT_FILE, fourcc, FPS, (WIDTH, HEIGHT)) # Generate static stars num_stars = 2000 stars = np.random.randint(0, 255, (num_stars, 3)) # x, y, brightness star_positions = np.random.rand(num_stars, 2) * [WIDTH, HEIGHT] star_speeds = np.random.rand(num_stars) * 2 + 0.5 # Parallax speed print(f"Generating {DURATION_SECONDS}s 4K video at {FPS} FPS...") for frame in range(FPS * DURATION_SECONDS): # Create black background img = np.zeros((HEIGHT, WIDTH, 3), dtype=np.uint8) # 1. Draw Moving Stars (Parallax effect moving downwards/outwards) for i in range(num_stars): x, y = star_positions[i] brightness = stars[i][2] # Move stars slightly to simulate drift y += star_speeds[i] * (frame / 10) if y > HEIGHT: y = 0 star_positions[i][1] = y cv2.circle(img, (int(x), int(y)), 1, (int(brightness), int(brightness), int(brightness)), -1) # 2. Draw Black Hole Center center = (WIDTH // 2, HEIGHT // 2) event_horizon_radius = 300 # Draw Event Horizon (Pure Black) cv2.circle(img, center, event_horizon_radius, (0, 0, 0), -1) # 3. Draw Rotating Accretion Disk # We simulate rotation by shifting the phase of the sine waves used for the disk texture rotation_angle = (frame / (FPS * DURATION_SECONDS)) * 2 * math.pi # Create a meshgrid for the disk area to apply gravitational lensing effect # Note: For performance in pure Python/OpenCV, we draw elliptical arcs for r in range(event_horizon_radius, event_horizon_radius + 400, 2): # Color gradient from bright orange/white to dark red intensity = max(0, 255 - (r - event_horizon_radius) * 0.8) color = (0, int(intensity * 0.5), int(intensity)) # BGR: Orange/Red # Draw rotating elliptical rings to simulate 3D disk angle_offset = math.sin(rotation_angle + r * 0.01) * 20 axes = (r + int(angle_offset), int(r * 0.3)) # Flattened ellipse # Add some noise/turbulence to the ring thickness = np.random.randint(1, 4) cv2.ellipse(img, center, axes, 0, 0, 360, color, thickness, cv2.LINE_AA) # Gravitational Lensing (Light bending over the top) if r < event_horizon_radius + 150: cv2.ellipse(img, center, (r, int(r * 0.8)), 0, 180, 360, (100, 150, 255), 1, cv2.LINE_AA) out.write(img) if frame % 60 == 0: print(f"Progress: {int((frame / (FPS * DURATION_SECONDS)) * 100)}%") out.release() print(f"Video saved successfully as {OUTPUT_FILE}")
